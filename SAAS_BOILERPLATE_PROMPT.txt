# SaaS Boilerplate Application - Complete Specification

Create a production-ready SaaS boilerplate application with the following modern tech stack and comprehensive testing coverage:

## Core Technology Stack

### Frontend
- **Next.js 14+** (App Router with TypeScript)
- **React 18+** with Server/Client Components
- **Material-UI v5 (MUI)** with Emotion styling
- **React Query (TanStack Query)** for state management and caching
- **SCSS** for custom styling
- **Responsive design** with mobile-first approach
- **Zustand for state management

### Backend
- **Next.js API Routes** (App Router)
- **Prisma ORM** with PostgreSQL database
- **Firebase Authentication** (email/password, Google OAuth)
- **Firebase Admin SDK** for server-side auth verification
- **Middleware-based** authentication and rate limiting

### Payment & Subscriptions
- **Stripe integration** with webhook handling
- **Subscription tiers** (Free, Pro, Enterprise)
- **Customer portal** for subscription management
- **Usage tracking** and billing

### Testing & Quality Assurance
- **Jest** for unit and integration testing
- **React Testing Library** for component testing
- **BDD approach** with Behavior-Driven Development patterns
- **Test coverage** minimum 80% for business logic
- **MSW (Mock Service Worker)** for API mocking
- **Supertest** for API endpoint testing
- **Playwright** for E2E testing (optional)

### Deployment
- **Vercel-optimized** configuration
- **Edge runtime** support where applicable
- **Environment variable** management
- **CI/CD pipeline** with automated testing

## Testing Strategy & BDD Implementation

### Test Structure (BDD Pattern)

Each business logic module should have comprehensive test coverage following BDD:

```typescript
// Example: src/__tests__/unit/utils/validation.test.ts
import { describe, it, expect } from '@jest/globals';
import {
  validateEmail,
  validatePassword,
  validateSymbol,
  validateInterval
} from '@/lib/utils/validation';

describe('Validation Utils', () => {
  describe('Email Validation', () => {
    describe('Given a valid email address', () => {
      it('should return true for standard email format', () => {
        expect(validateEmail('user@example.com')).toBe(true);
      });

      it('should return true for email with subdomains', () => {
        expect(validateEmail('user@mail.example.com')).toBe(true);
      });

      it('should return true for email with plus addressing', () => {
        expect(validateEmail('user+tag@example.com')).toBe(true);
      });
    });

    describe('Given an invalid email address', () => {
      it('should return false for missing @ symbol', () => {
        expect(validateEmail('userexample.com')).toBe(false);
      });

      it('should return false for missing domain', () => {
        expect(validateEmail('user@')).toBe(false);
      });

      it('should return false for spaces in email', () => {
        expect(validateEmail('user @example.com')).toBe(false);
      });

      it('should return false for empty string', () => {
        expect(validateEmail('')).toBe(false);
      });
    });
  });

  describe('Password Validation', () => {
    describe('Given a valid password', () => {
      it('should accept password with 6 characters', () => {
        const result = validatePassword('pass12');
        expect(result.valid).toBe(true);
        expect(result.message).toBeUndefined();
      });

      it('should accept password with 20 characters', () => {
        const result = validatePassword('a'.repeat(20));
        expect(result.valid).toBe(true);
      });
    });

    describe('Given an invalid password', () => {
      it('should reject password with less than 6 characters', () => {
        const result = validatePassword('pass1');
        expect(result.valid).toBe(false);
        expect(result.message).toBe('Password must be at least 6 characters long');
      });

      it('should reject empty password', () => {
        const result = validatePassword('');
        expect(result.valid).toBe(false);
      });
    });
  });

  describe('Symbol Validation', () => {
    describe('Given valid trading symbols', () => {
      it('should accept uppercase symbol', () => {
        expect(validateSymbol('AAPL')).toBe(true);
      });

      it('should accept symbol with exchange prefix', () => {
        expect(validateSymbol('NASDAQ:AAPL')).toBe(true);
      });

      it('should accept symbol with underscore', () => {
        expect(validateSymbol('BTC_USD')).toBe(true);
      });
    });

    describe('Given invalid trading symbols', () => {
      it('should reject empty string', () => {
        expect(validateSymbol('')).toBe(false);
      });

      it('should reject symbol with special characters', () => {
        expect(validateSymbol('AAPL@#')).toBe(false);
      });

      it('should reject symbol with spaces', () => {
        expect(validateSymbol('AAPL STOCK')).toBe(false);
      });
    });
  });
});
```

```typescript
// Example: src/__tests__/unit/services/auth.service.test.ts
import { describe, it, expect, jest, beforeEach, afterEach } from '@jest/globals';
import { authenticateRequest } from '@/lib/middleware/auth';
import { NextRequest } from 'next/server';
import * as firebaseAdmin from '@/lib/firebase/adminApp';
import { prisma } from '@/lib/prisma';

// Mock dependencies
jest.mock('@/lib/firebase/adminApp');
jest.mock('@/lib/prisma', () => ({
  prisma: {
    user: {
      findUnique: jest.fn(),
      create: jest.fn(),
    },
  },
}));

describe('Authentication Service', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  describe('authenticateRequest', () => {
    describe('Given a valid authentication token', () => {
      it('should return authenticated user when user exists', async () => {
        // Arrange
        const mockToken = 'valid-token';
        const mockUid = 'user-123';
        const mockEmail = 'user@example.com';
        const mockUserId = 'db-user-123';

        const mockRequest = new NextRequest('http://localhost:3000/api/test', {
          headers: { authorization: `Bearer ${mockToken}` },
        });

        (firebaseAdmin.verifyIdToken as jest.Mock).mockResolvedValue({
          uid: mockUid,
          email: mockEmail,
        });

        (prisma.user.findUnique as jest.Mock).mockResolvedValue({
          id: mockUserId,
          firebaseUid: mockUid,
          email: mockEmail,
        });

        // Act
        const result = await authenticateRequest(mockRequest);

        // Assert
        expect(result).toEqual({
          uid: mockUid,
          email: mockEmail,
          userId: mockUserId,
        });
        expect(firebaseAdmin.verifyIdToken).toHaveBeenCalledWith(mockToken);
        expect(prisma.user.findUnique).toHaveBeenCalledWith({
          where: { firebaseUid: mockUid },
        });
      });

      it('should create new user when user does not exist', async () => {
        // Arrange
        const mockToken = 'valid-token';
        const mockUid = 'new-user-123';
        const mockEmail = 'newuser@example.com';
        const mockUserId = 'new-db-user-123';

        const mockRequest = new NextRequest('http://localhost:3000/api/test', {
          headers: { authorization: `Bearer ${mockToken}` },
        });

        (firebaseAdmin.verifyIdToken as jest.Mock).mockResolvedValue({
          uid: mockUid,
          email: mockEmail,
        });

        (prisma.user.findUnique as jest.Mock).mockResolvedValue(null);
        (prisma.user.create as jest.Mock).mockResolvedValue({
          id: mockUserId,
          firebaseUid: mockUid,
          email: mockEmail,
        });

        // Act
        const result = await authenticateRequest(mockRequest);

        // Assert
        expect(result.userId).toBe(mockUserId);
        expect(prisma.user.create).toHaveBeenCalledWith({
          data: {
            firebaseUid: mockUid,
            email: mockEmail,
          },
        });
      });
    });

    describe('Given an invalid authentication token', () => {
      it('should throw error when authorization header is missing', async () => {
        // Arrange
        const mockRequest = new NextRequest('http://localhost:3000/api/test');

        // Act & Assert
        await expect(authenticateRequest(mockRequest)).rejects.toThrow(
          'Missing or invalid authorization header'
        );
      });

      it('should throw error when token format is invalid', async () => {
        // Arrange
        const mockRequest = new NextRequest('http://localhost:3000/api/test', {
          headers: { authorization: 'InvalidFormat token' },
        });

        // Act & Assert
        await expect(authenticateRequest(mockRequest)).rejects.toThrow(
          'Missing or invalid authorization header'
        );
      });

      it('should throw error when token verification fails', async () => {
        // Arrange
        const mockRequest = new NextRequest('http://localhost:3000/api/test', {
          headers: { authorization: 'Bearer invalid-token' },
        });

        (firebaseAdmin.verifyIdToken as jest.Mock).mockRejectedValue(
          new Error('Token verification failed')
        );

        // Act & Assert
        await expect(authenticateRequest(mockRequest)).rejects.toThrow(
          'Token verification failed'
        );
      });

      it('should throw error when token payload is incomplete', async () => {
        // Arrange
        const mockRequest = new NextRequest('http://localhost:3000/api/test', {
          headers: { authorization: 'Bearer valid-token' },
        });

        (firebaseAdmin.verifyIdToken as jest.Mock).mockResolvedValue({
          uid: 'user-123',
          // Missing email
        });

        // Act & Assert
        await expect(authenticateRequest(mockRequest)).rejects.toThrow(
          'Invalid token payload'
        );
      });
    });
  });
});
```

### API Route Tests (Integration)

```typescript
// Example: src/__tests__/integration/api/user.test.ts
import { describe, it, expect, beforeAll, afterAll } from '@jest/globals';
import { NextRequest } from 'next/server';
import { GET, PUT } from '@/app/api/user/profile/route';
import { prisma } from '@/lib/prisma';
import * as auth from '@/lib/middleware/auth';

// Mock authentication
jest.mock('@/lib/middleware/auth');

describe('User Profile API', () => {
  const mockUserId = 'test-user-123';
  const mockAuthUser = {
    uid: 'firebase-uid-123',
    email: 'test@example.com',
    userId: mockUserId,
  };

  beforeAll(async () => {
    // Setup test database
    await prisma.user.create({
      data: {
        id: mockUserId,
        firebaseUid: 'firebase-uid-123',
        email: 'test@example.com',
      },
    });
  });

  afterAll(async () => {
    // Cleanup
    await prisma.user.delete({ where: { id: mockUserId } });
  });

  describe('GET /api/user/profile', () => {
    describe('When user is authenticated', () => {
      it('should return user profile data', async () => {
        // Arrange
        (auth.authenticateRequest as jest.Mock).mockResolvedValue(mockAuthUser);
        const request = new NextRequest('http://localhost:3000/api/user/profile');

        // Act
        const response = await GET(request);
        const data = await response.json();

        // Assert
        expect(response.status).toBe(200);
        expect(data.user).toMatchObject({
          id: mockUserId,
          email: 'test@example.com',
        });
      });
    });

    describe('When user is not authenticated', () => {
      it('should return 401 unauthorized', async () => {
        // Arrange
        (auth.authenticateRequest as jest.Mock).mockRejectedValue(
          new Error('Unauthorized')
        );
        const request = new NextRequest('http://localhost:3000/api/user/profile');

        // Act
        const response = await GET(request);

        // Assert
        expect(response.status).toBe(401);
      });
    });
  });

  describe('PUT /api/user/profile', () => {
    describe('When updating valid profile data', () => {
      it('should update and return updated profile', async () => {
        // Arrange
        (auth.authenticateRequest as jest.Mock).mockResolvedValue(mockAuthUser);
        const updateData = {
          displayName: 'Updated Name',
          bio: 'Updated bio',
        };
        const request = new NextRequest('http://localhost:3000/api/user/profile', {
          method: 'PUT',
          body: JSON.stringify(updateData),
        });

        // Act
        const response = await PUT(request);
        const data = await response.json();

        // Assert
        expect(response.status).toBe(200);
        expect(data.user.displayName).toBe('Updated Name');
        expect(data.user.bio).toBe('Updated bio');
      });
    });

    describe('When updating with invalid data', () => {
      it('should return 400 for invalid email format', async () => {
        // Arrange
        (auth.authenticateRequest as jest.Mock).mockResolvedValue(mockAuthUser);
        const invalidData = {
          email: 'invalid-email',
        };
        const request = new NextRequest('http://localhost:3000/api/user/profile', {
          method: 'PUT',
          body: JSON.stringify(invalidData),
        });

        // Act
        const response = await PUT(request);

        // Assert
        expect(response.status).toBe(400);
      });
    });
  });
});
```

### Admin Middleware & Security

**adminAuth Middleware:**
```typescript
// lib/middleware/adminAuth.ts
export async function requireAdmin(request: NextRequest) {
  const user = await authenticateRequest(request);
  
  const adminUser = await prisma.adminUser.findUnique({
    where: { userId: user.userId },
  });
  
  if (!adminUser) {
    throw new Error('Unauthorized: Admin access required');
  }
  
  return { ...user, adminRole: adminUser.role, permissions: adminUser.permissions };
}

export function requirePermission(permission: string) {
  return async (request: NextRequest) => {
    const admin = await requireAdmin(request);
    
    if (!admin.permissions.includes(permission) && admin.adminRole !== 'super_admin') {
      throw new Error(`Unauthorized: Missing permission '${permission}'`);
    }
    
    return admin;
  };
}
```

```typescript
// lib/middleware/adminAuth.ts
import { NextRequest } from 'next/server';
import { authenticateRequest } from './auth';
import { prisma } from '../prisma';

export async function requireAdmin(request: NextRequest) {
  const user = await authenticateRequest(request);
  
  const adminUser = await prisma.adminUser.findUnique({
    where: { userId: user.userId },
    include: { user: true },
  });
  
  if (!adminUser) {
    throw new Error('Unauthorized: Admin access required');
  }
  
  return {
    ...user,
    adminRole: adminUser.role,
    permissions: adminUser.permissions as string[],
  };
}

export function hasPermission(
  userPermissions: string[],
  role: string,
  requiredPermission: string
): boolean {
  if (role === 'super_admin') return true;
  return userPermissions.includes(requiredPermission);
}
```

### Admin Service Example
```typescript
// lib/services/admin/user-management.service.ts
import { prisma } from '@/lib/prisma';
import { logAdminAction } from './audit-log.service';

export async function listUsers(params: {
  page: number;
  limit: number;
  search?: string;
  tier?: string;
  status?: string;
}) {
  const { page, limit, search, tier, status } = params;
  const skip = (page - 1) * limit;

  const where: any = {};
  
  if (search) {
    where.OR = [
      { email: { contains: search, mode: 'insensitive' } },
      { displayName: { contains: search, mode: 'insensitive' } },
    ];
  }
  
  if (tier) {
    where.subscriptionTier = tier;
  }
  
  if (status) {
    where.subscriptionStatus = status;
  }

  const [users, total] = await Promise.all([
    prisma.user.findMany({
      where,
      skip,
      take: limit,
      orderBy: { createdAt: 'desc' },
      include: {
        settings: true,
        _count: {
          select: { loginHistory: true },
        },
      },
    }),
    prisma.user.count({ where }),
  ]);

  return {
    users,
    total,
    pages: Math.ceil(total / limit),
    currentPage: page,
  };
}

export async function suspendUser(
  userId: string,
  adminId: string,
  reason: string
) {
  const user = await prisma.user.update({
    where: { id: userId },
    data: {
      subscriptionStatus: 'suspended',
    },
  });

  await logAdminAction({
    userId: adminId,
    action: 'user.suspended',
    resource: 'User',
    resourceId: userId,
    metadata: { reason },
  });

  return user;
}
```

### Admin API Route Example
```typescript
// app/api/admin/users/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { requireAdmin } from '@/lib/middleware/adminAuth';
import { listUsers } from '@/lib/services/admin/user-management.service';

export async function GET(request: NextRequest) {
  try {
    await requireAdmin(request);

    const searchParams = request.nextUrl.searchParams;
    const page = parseInt(searchParams.get('page') || '1');
    const limit = parseInt(searchParams.get('limit') || '20');
    const search = searchParams.get('search') || undefined;
    const tier = searchParams.get('tier') || undefined;
    const status = searchParams.get('status') || undefined;

    const result = await listUsers({ page, limit, search, tier, status });

    return NextResponse.json(result);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message },
      { status: error.message.includes('Unauthorized') ? 403 : 500 }
    );
  }
}
```

### Admin Component Example
```typescript
// components/admin/UserTable.tsx
'use client';

import { useState } from 'react';
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableRow,
  IconButton,
  Menu,
  MenuItem,
  Chip,
  Avatar,
} from '@mui/material';
import { MoreVert, Block, Edit, Delete } from '@mui/icons-material';

interface User {
  id: string;
  email: string;
  displayName?: string;
  photoURL?: string;
  subscriptionTier: string;
  subscriptionStatus: string;
  createdAt: string;
}

export default function UserTable({ users }: { users: User[] }) {
  const [anchorEl, setAnchorEl] = useState<null | HTMLElement>(null);
  const [selectedUser, setSelectedUser] = useState<User | null>(null);

  const handleMenuOpen = (event: React.MouseEvent<HTMLElement>, user: User) => {
    setAnchorEl(event.currentTarget);
    setSelectedUser(user);
  };

  const handleMenuClose = () => {
    setAnchorEl(null);
    setSelectedUser(null);
  };

  const getTierColor = (tier: string) => {
    switch (tier) {
      case 'free': return 'default';
      case 'pro': return 'primary';
      case 'enterprise': return 'secondary';
      default: return 'default';
    }
  };

  return (
    <>
      <Table>
        <TableHead>
          <TableRow>
            <TableCell>User</TableCell>
            <TableCell>Email</TableCell>
            <TableCell>Tier</TableCell>
            <TableCell>Status</TableCell>
            <TableCell>Joined</TableCell>
            <TableCell align="right">Actions</TableCell>
          </TableRow>
        </TableHead>
        <TableBody>
          {users.map((user) => (
            <TableRow key={user.id}>
              <TableCell>
                <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
                  <Avatar src={user.photoURL} alt={user.displayName}>
                    {user.displayName?.[0] || user.email[0]}
                  </Avatar>
                  {user.displayName || 'No name'}
                </div>
              </TableCell>
              <TableCell>{user.email}</TableCell>
              <TableCell>
                <Chip
                  label={user.subscriptionTier}
                  color={getTierColor(user.subscriptionTier)}
                  size="small"
                />
              </TableCell>
              <TableCell>
                <Chip
                  label={user.subscriptionStatus}
                  color={user.subscriptionStatus === 'active' ? 'success' : 'default'}
                  size="small"
                />
              </TableCell>
              <TableCell>
                {new Date(user.createdAt).toLocaleDateString()}
              </TableCell>
              <TableCell align="right">
                <IconButton
                  size="small"
                  onClick={(e) => handleMenuOpen(e, user)}
                >
                  <MoreVert />
                </IconButton>
              </TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>

      <Menu
        anchorEl={anchorEl}
        open={Boolean(anchorEl)}
        onClose={handleMenuClose}
      >
        <MenuItem onClick={handleMenuClose}>
          <Edit fontSize="small" sx={{ mr: 1 }} />
          Edit User
        </MenuItem>
        <MenuItem onClick={handleMenuClose}>
          <Block fontSize="small" sx={{ mr: 1 }} />
          Suspend Account
        </MenuItem>
        <MenuItem onClick={handleMenuClose} sx={{ color: 'error.main' }}>
          <Delete fontSize="small" sx={{ mr: 1 }} />
          Delete User
        </MenuItem>
      </Menu>
    </>
  );
}
```
